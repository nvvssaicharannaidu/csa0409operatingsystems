#include <stdio.h>

int main() {
    int bt[10], rem[10], n, tq;
    int i, done;

    printf("Enter number of processes: ");
    scanf("%d", &n);

    for(i = 0; i < n; i++) {
        printf("Burst Time of P%d: ", i + 1);
        scanf("%d", &bt[i]);
        rem[i] = bt[i];
    }

    printf("Enter Time Quantum: ");
    scanf("%d", &tq);

    printf("\nExecution Order:\n");

    do {
        done = 1;
        for(i = 0; i < n; i++) {
            if(rem[i] > 0) {
                done = 0;
                printf("P%d ", i + 1);

                if(rem[i] > tq)
                    rem[i] -= tq;
                else
                    rem[i] = 0;
            }
        }
    } while(!done);

    return 0;
}
